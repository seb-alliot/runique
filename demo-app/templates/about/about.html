// After an action: add + redirect
pub async fn create_post(
    mut request: Request,
) -> AppResult<Response> {
    let form: BlogForm = request.form("blog");
    let form = match ValidationForm::try_form(form, &request).await {
        Ok(validated) => validated,
        Err(form) => {
            context_update!(request => { "blog_form" => &form });
            return request.render("blog/blog.html");
        }
    };
    if form.save(&request.engine.db).await.is_ok() {
        success!(request.notices => "Article published!");
    }
    Ok(Redirect::to("/blog").into_response())
}
