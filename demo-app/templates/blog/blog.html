#[form(schema = blog,
       fields = [title, email, summary, website, content])]
pub struct BlogForm;

#[async_trait]
impl RuniqueForm for BlogForm {
    async fn clean(&mut self) -> Result<(), StrMap> {
        let mut errors = StrMap::new();
        if self.cleaned::<String>("title").unwrap_or_default().len() < 3 {
            errors.insert("title".into(), "3 characters minimum".into());
        }
        if !self.cleaned::<String>("email").unwrap_or_default().contains('@') {
            errors.insert("email".into(), "Invalid address".into());
        }
        if errors.is_empty() { Ok(()) }
        else { Err(errors) }
    }
}

pub async fn blog_submit(
    mut request: Request,
) -> AppResult<Response> {
    let template = "blog/blog.html";
    let form: BlogForm = request.form("blog");
    let form = match ValidationForm::try_form(form, &request).await {
        Ok(validated) => validated,
        // GET, or sent with errors: the form again
        Err(form) => {
            context_update!(request => { "blog_form" => &form });
            return request.render(template);
        }
    };
    match form.save(&request.engine.db).await {
        Ok(_) => {
            success!(request.notices => "Article published!");
            Ok(Redirect::to("/blog").into_response())
        }
        // Refused by the database: the reason is on the form
        Err(form) => {
            context_update!(request => { "blog_form" => &form });
            request.render(template)
        }
    }
}
