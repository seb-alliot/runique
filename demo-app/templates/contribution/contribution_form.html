#[form(schema = contribution,
       fields = [contribution_type, title, content])]
pub struct ContributionForm;

#[async_trait]
impl RuniqueForm for ContributionForm {
    async fn clean(&mut self) -> Result<(), StrMap> {
        let mut errors = StrMap::new();
        if self.cleaned::<String>("title").unwrap_or_default().len() < 3 {
            errors.insert("title".into(), "3 characters minimum".into());
        }
        if self.cleaned::<String>("content").unwrap_or_default().len() < 10 {
            errors.insert("content".into(), "10 characters minimum".into());
        }
        if errors.is_empty() { Ok(()) }
        else { Err(errors) }
    }
}

pub async fn contribution_submit(
    mut request: Request,
) -> AppResult<Response> {
    let template = "contribution/contribution_form.html";
    let form: ContributionForm = request.form();
    // Protected route — redirects if not signed in
    let Some(user_id) = request.user.as_ref().map(|user| user.id) else {
        return Ok(Redirect::to("/login").into_response());
    };
    let mut form = match ValidationForm::try_form(form, &request, "contribution").await {
        Ok(validated) => validated.into_form(),
        Err(form) => {
            context_update!(request => { "contribution_form" => &form });
            return request.render(template);
        }
    };
    // Own save: it sets the owner, which the form doesn't carry
    match save_contribution(&mut form, &request.engine.db, user_id).await {
        Ok(_) => {
            success!(request.notices => "Contribution saved!");
            Ok(Redirect::to("/contributions").into_response())
        }
        Err(err) => {
            form.forms_mut().database_error(&err);
            context_update!(request => { "contribution_form" => &form });
            request.render(template)
        }
    }
}
